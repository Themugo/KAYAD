==================== ACCOUNT CAPABILITIES ====================
// ONE ACCOUNT, MANY CAPABILITIES. A read-only view over the existing authorities - it never grants, changes or infers anything.
//   buyer            : every active registered account.
//   privateSeller    : policy-controlled. Default mode is free/open with no seller approval; suspension/revocation remain enforceable. When approval is enabled, capability status is authoritative.
//   dealer           : users.role === "dealer" (set only by administrator approval); the dealers/dealer_verifications rows are the
//                      application, separate from the identity.
//   inspector        : users.role === "ghost_checker" (set only by administrator approval); InspectorApplication is the application.
// A pending or rejected business application never reduces the buyer or private-seller entries.
import { findOne } from "../db/index.js";
import InspectorApplication from "../models/InspectorApplication.js";
import { resolveSellerState } from "./privateSellerCapability.service.js";

const BUSINESS_STATUS = (granted, application) => (granted ? "approved" : application || "none");

export async function getAccountCapabilities(user) {
  const userId = String(user.id ?? user._id);
  const isStaff = !["user", "dealer", "ghost_checker", "individual_seller"].includes(user.role);
  const seller = await resolveSellerState({ id: userId, role: user.role, status: user.status });
  // Legacy individual_seller identities were marked pending solely by the old seller-approval workflow.
  // Under the free default policy they remain active personal accounts; the forward migration also normalizes
  // this persisted status so routes using requireApproved see the same result after rollout.
  const active = (user.status || "approved") === "approved"
    || (user.role === "individual_seller" && seller.source === "policy_default" && user.status === "pending");
  const dealerRow = user.role === "dealer" ? null : await findOne("dealers", { user: userId });
  const dealerVerification = dealerRow ? await findOne("dealer_verifications", { user: userId }) : null;
  const inspectorApps = await InspectorApplication.find({ user: userId }).sort({ createdAt: -1 }).lean();
  const inspectorApp = inspectorApps?.[0]?.status || null;

  const dealerApplication = dealerRow ? (dealerVerification?.verificationStatus ?? "draft") : null;
  return {
    userId,
    identity: { role: user.role, status: user.status || "approved", emailVerified: user.emailVerified === true },
    buyer: { granted: !isStaff && active },
    privateSeller: {
      status: seller.status, // none | pending | approved | rejected | suspended | revoked | not_applicable
      granted: seller.status === "approved",
      canApply: user.role === "user" && ["none", "rejected"].includes(seller.status),
    },
    dealer: {
      granted: user.role === "dealer",
      status: BUSINESS_STATUS(user.role === "dealer" && !dealerApplication, dealerApplication),
      canApply: user.role === "user" && !dealerRow,
    },
    inspector: {
      granted: user.role === "ghost_checker",
      status: BUSINESS_STATUS(user.role === "ghost_checker", inspectorApp),
      canApply: user.role === "user" && inspectorApp !== "pending",
    },
  };
}

// Inside a mounted Express router req.path is relative to the mount point
// (POST /api/cars arrives as "/"), so comparing it to "/cars" never matched and
// progressive access was unreachable. Use the full mounted path instead.
export const isListingCreationRequest = (req) => {
  if (req.method !== "POST") return false;
  const full = String(`${req.baseUrl || ""}${req.path || ""}` || req.originalUrl || "")
    .split("?")[0]
    .replace(/\/+$/, "");
  return /\/(cars|add-car|inventory)$/.test(full);
};

// 🔒 REQUIRE DEALER VERIFICATION
// =============================
// Checks if dealer is verified before allowing protected operations
// Applied to: listing creation, auction start, escrow initiation
export const requireDealerVerification = async (req, res, next) => {
  try {
    const userId = req.user.id;

    // Check if user is a dealer
    let dealer = await Dealer.findOne({ user: userId });

    // PHASE 13D (DI-4): a private seller (individual_seller) listing their own vehicle is NOT a dealership. This
    // middleware used to INSERT an `approved: true` Dealer row for any individual_seller that reached it, whatever
    // their users.status, which silently created (and auto-approved) a dealership identity from a personal listing.
    // It no longer creates anything: a private seller with no Dealer row is let through to the listing handler (which
    // keeps its own seller email-verification / publication-review rules) with `req.dealer = null`, exactly the
    // authorization outcome the old auto-created row produced, minus the dealership record. Dealer rows that already
    // exist are neither modified nor removed here. Legitimate dealer onboarding (role `dealer`, trigger-created row,
    // verification workflow) is unchanged.
    // 13E: a personal account (role user) with an APPROVED private-seller capability (set by sellerListingAccess) is also a private seller.
    const isPrivateSeller = req.user.role === "individual_seller" || (req.user.role === "user" && req.sellerCapability?.status === "approved");

    if (!dealer && !isPrivateSeller) {
      return res.status(403).json({
        success: false,
        message: "Dealer profile not found",
        requiresAction: "create_dealer_profile",
      });
    }

    // A legacy approved flag must never override a current suspension or rejection.
    // Resolve the verification record before applying backward-compatible approval.
    const verification = await DealerVerification.findOne({ user: userId });

    if (verification?.verificationStatus === "suspended") {
      return res.status(403).json({
        success: false,
        code: "DEALER_ACCOUNT_SUSPENDED",
        message: "Dealer account is suspended and requires an explicit admin reinstatement.",
        verificationStatus: "suspended",
        suspensionReason: verification.suspensionReason || dealer?.suspensionReason || null,
        suspensionExpiresAt: verification.suspensionExpiresAt || null,
      });
    }

    if (verification?.verificationStatus === "rejected") {
      return res.status(403).json({
        success: false,
        code: "DEALER_VERIFICATION_REJECTED",
        message: "Dealer verification was rejected. Update the application and resubmit it.",
        verificationStatus: "rejected",
        requiresAction: "resubmit_verification",
        rejectionReason: verification.rejectionReason || null,
      });
    }

    // PHASE 13D: private seller without a Dealer row -> no dealership identity, listing flow continues.
    if (!dealer) {
      req.dealer = null;
      req.verification = verification || null;
      return next();
    }

    // Backward compatibility applies only when there is no newer blocking decision.
    if (dealer.approved === true && (!verification || verification.verificationStatus === "approved")) {
      req.dealer = dealer;
      req.verification = verification || null;
      return next();
    }

    if (!verification) {
      return res.status(403).json({
        success: false,
        message: "Dealer verification required",
        verificationStatus: "none",
        requiresAction: "submit_verification",
        verificationUrl: "/api/verification/submit",
      });
    }

    // Check verification status — progressive access for pending/under_review
    if (verification.verificationStatus !== "approved") {
      // STAGE 5: progressive (pre-approval) listing access exists ONLY in the owner's
      // `self_onboarding` mode. In `approval_required`/`closed` — and whenever the policy
      // cannot be trusted — an unapproved dealer cannot create listings at all.
      let progressiveAllowed = false;
      if (["pending", "under_review"].includes(verification.verificationStatus) && isListingCreationRequest(req)) {
        const onboarding = await getOnboardingPolicy();
        progressiveAllowed =
          (onboarding.source === "default" || onboarding.source === "database") &&
          onboarding.policy.dealerOnboardingMode === "self_onboarding";
      }
      // Progressive verification: allow listing creation if < 3 cars and not rejected
      if (progressiveAllowed) {
        const carCount = await Car.countDocuments({ dealer: userId });
        if (carCount < 3) {
          req.dealer = dealer;
          req.verification = verification;
          req.progressiveAccess = true;
          return next();
        }
      }

      return res.status(403).json({
        success: false,
        message: getVerificationMessage(verification.verificationStatus),
        code: ["pending", "under_review"].includes(verification.verificationStatus) ? "DEALER_APPROVAL_PENDING" : undefined,
        verificationStatus: verification.verificationStatus,
        requiresAction: getRequiredAction(verification.verificationStatus),
        verificationUrl: "/api/verification/status",
        rejectionReason: verification.rejectionReason,
        progressiveLimit: 3,
        progressiveUsed: verification.verificationStatus === "pending" || verification.verificationStatus === "under_review",
      });
    }

    // Verification passed
    req.dealer = dealer;
    req.verification = verification;
    next();
  } catch (err) {
    logError("Dealer verification middleware error", err);
    return res.status(500).json({
      success: false,
      message: "Verification check failed",
    });
  }
};

// 🔒 REQUIRE DEALER VERIFICATION (WARN ONLY)

// Logs verification status but doesn't block requests
// Used for gradual rollout and monitoring
export const requireDealerVerificationWarn = async (req, res, next) => {
  try {
    const userId = req.user.id;

    const dealer = await Dealer.findOne({ user: userId });
    if (!dealer) {
      logWarn("Dealer profile not found", { userId });
      req.dealer = dealer;
      return next();
    }

    if (dealer.approved === true) {
      req.dealer = dealer;
      return next();
    }

    const verification = await DealerVerification.findOne({ user: userId });

    if (!verification) {
      logWarn("Dealer verification not submitted", { userId, dealerId: dealer._id });
      req.dealer = dealer;
      return next();
    }

    if (verification.verificationStatus !== "approved") {
      logWarn("Dealer not verified", {
        userId,
        dealerId: dealer._id,
        verificationStatus: verification.verificationStatus,
        path: req.path,
      });
    }

    req.dealer = dealer;
    req.verification = verification;
    next();
  } catch (err) {
    logError("Dealer verification warn middleware error", err);
    req.dealer = null;
    next();
  }
};

// Helper function to check verification status without blocking
export const checkDealerVerificationStatus = async (userId) => {
  try {
    const dealer = await Dealer.findOne({ user: userId });
    if (!dealer) return { verified: false, reason: "no_dealer_profile" };

    const verification = await DealerVerification.findOne({ user: userId });

    if (verification?.verificationStatus === "suspended") {
      return { verified: false, reason: "suspended", suspensionReason: verification.suspensionReason || dealer.suspensionReason || null };
    }
    if (verification?.verificationStatus === "rejected") {
      return { verified: false, reason: "rejected", rejectionReason: verification.rejectionReason || null };
    }

    // Legacy approval is only a fallback when no newer blocking decision exists.
    if (dealer.approved === true && (!verification || verification.verificationStatus === "approved")) {
      return { verified: true, reason: verification ? "approved" : "legacy_approval" };
    }

    if (!verification) return { verified: false, reason: "no_verification" };

    if (verification.verificationStatus === "approved") {
      return { verified: true, reason: "approved" };
    }

    return {
      verified: false,
      reason: verification.verificationStatus,
      verificationStatus: verification.verificationStatus,
      rejectionReason: verification.rejectionReason,
    };
  } catch (err) {
    logError("Check dealer verification status error", err, { userId });
    return { verified: false, reason: "error" };
  }
};

==================== ONBOARDING POLICY ====================
// backend/services/onboardingPolicy.service.js
// ─────────────────────────────────────────────────────────────────────────
// Stage 5 (2026-10-19) — OWNER-CONTROLLED ONBOARDING POLICY.
//
// Storage: the established key/value control plane, public.system_settings
// (key = 'onboarding_policy', value = jsonb). The same table already holds
// 'system_status' (kill-switch). NO schema change is needed, so no migration is
// required to launch; the row is created by the first authorized save.
//
// Policy fields (all server-validated enums; unknown keys are rejected):
//   privateSellerRegistration  enabled | disabled
//   privateSellerApprovalMode  disabled | required (defaults to disabled)
//   privateSellerPackageMode   free (paid modes require a later package catalogue + verified checkout integration)
//   dealerOnboardingMode       self_onboarding | approval_required | closed
//   listingModeration          hold_for_review | auto_publish
//   dealerDecisionAuthority    owner_only | admins     (who may approve/reject dealers & sellers)
//   listingDecisionAuthority   owner_only | admins     (who may approve/reject listings)
// Email verification for seller actions is NOT a setting: it is always required.
//
// Fail-safe contract (never "error ⇒ allowed"):
//   row missing                → INITIAL_DEFAULTS (deliberate launch configuration), source "default"
//   store unreachable / error  → RESTRICTIVE values, source "unavailable"
//   row or a field is corrupt  → RESTRICTIVE value for the corrupt field(s), source "invalid"
// Reads are cached for CACHE_TTL_MS (a change reaches every instance within that
// time); the cache is dropped on every successful write by this instance.
// Writes are compare-and-swap on the row's revision AND updated_at, so two
// concurrent editors cannot silently overwrite each other.
// ─────────────────────────────────────────────────────────────────────────
import * as supabaseModule from "../utils/supabase.js";
import * as loggerModule from "../infrastructure/logging/index.js";

// Namespace imports + guarded calls: this service is imported by many modules, and some test doubles
// replace utils/supabase.js or utils/logger.js with partial factories. A missing helper must degrade to
// the safe behaviour (treat as connected / skip logging), never to an import-time failure.
const getSupabase = () => supabaseModule.getSupabase();
const isSupabaseConnected = () =>
  typeof supabaseModule.isSupabaseConnected === "function" ? supabaseModule.isSupabaseConnected() : true;
const logError = (...a) => { try { loggerModule.logError?.(...a); } catch { /* logging must not throw */ } };
const logWarn = (...a) => { try { loggerModule.logWarn?.(...a); } catch { /* logging must not throw */ } };

export const ONBOARDING_POLICY_KEY = "onboarding_policy";
export const SCHEMA_VERSION = 2;
const CACHE_TTL_MS = 5_000;

export const POLICY_FIELDS = Object.freeze({
  privateSellerRegistration: Object.freeze(["enabled", "disabled"]),
  privateSellerApprovalMode: Object.freeze(["disabled", "required"]),
  privateSellerPackageMode: Object.freeze(["free", "optional", "required"]),
  dealerOnboardingMode: Object.freeze(["self_onboarding", "approval_required", "closed"]),
  listingModeration: Object.freeze(["hold_for_review", "auto_publish"]),
  dealerDecisionAuthority: Object.freeze(["owner_only", "admins"]),
  listingDecisionAuthority: Object.freeze(["owner_only", "admins"]),
});

/** Deliberate initial configuration when no row exists. Dealers are NOT opened by default. */
export const INITIAL_DEFAULTS = Object.freeze({
  privateSellerRegistration: "enabled",
  privateSellerApprovalMode: "disabled",
  privateSellerPackageMode: "free",
  dealerOnboardingMode: "approval_required",
  listingModeration: "hold_for_review",
  dealerDecisionAuthority: "owner_only",
  listingDecisionAuthority: "owner_only",
});

/** Used whenever the real value cannot be trusted. Every value is the most restrictive one. */
export const RESTRICTIVE = Object.freeze({
  privateSellerRegistration: "disabled",
  privateSellerApprovalMode: "disabled",
  privateSellerPackageMode: "free",
  dealerOnboardingMode: "closed",
  listingModeration: "hold_for_review",
  dealerDecisionAuthority: "owner_only",
  listingDecisionAuthority: "owner_only",
});

export class OnboardingPolicyError extends Error {
  constructor(code, message, status, extra = {}) {
    super(message);
    this.name = "OnboardingPolicyError";
    this.code = code;
    this.status = status;
    Object.assign(this, extra);
  }
}

const isPlainObject = (v) => v !== null && typeof v === "object" && !Array.isArray(v);
const meta = (raw) => ({
  revision: Number.isInteger(raw?.revision) && raw.revision >= 0 ? raw.revision : 0,
  updatedAt: typeof raw?.updatedAt === "string" ? raw.updatedAt : null,
  updatedBy: typeof raw?.updatedBy === "string" ? raw.updatedBy : null,
});

/** Pure: turn a stored value into a policy. Corrupt fields become RESTRICTIVE and are reported. */
export function normalizePolicyValue(raw) {
  if (!isPlainObject(raw)) {
    return { policy: { ...RESTRICTIVE }, issues: ["value is not an object"], meta: meta(null) };
  }
  const policy = {};
  const issues = [];
  for (const [field, allowed] of Object.entries(POLICY_FIELDS)) {
    if (raw[field] === undefined) policy[field] = INITIAL_DEFAULTS[field]; // older revision without the field
    else if (typeof raw[field] === "string" && allowed.includes(raw[field])) policy[field] = raw[field];
    else { policy[field] = RESTRICTIVE[field]; issues.push(`${field} is invalid`); }
  }
  // STAGE 5A: a partially corrupt record must not display (or half-apply) a permissive value for a *different* field. Any issue
  // means the whole record is untrusted: every field becomes RESTRICTIVE until the owner saves a valid configuration.
  if (issues.length) return { policy: { ...RESTRICTIVE }, issues, meta: meta(raw) };
  return { policy, issues, meta: meta(raw) };
}

/** Pure: validate a change request. Returns { ok, errors, changes }. */
export function validatePolicyChanges(input) {
  const errors = [];
  if (!isPlainObject(input)) return { ok: false, errors: ["changes must be an object"], changes: {} };
  const changes = {};
  for (const [k, v] of Object.entries(input)) {
    if (!Object.hasOwn(POLICY_FIELDS, k)) { errors.push(`unknown setting: ${k}`); continue; }
    if (typeof v !== "string" || !POLICY_FIELDS[k].includes(v)) { errors.push(`${k} must be one of: ${POLICY_FIELDS[k].join(", ")}`); continue; }
    changes[k] = v;
  }
  if (!errors.length && !Object.keys(changes).length) errors.push("no changes supplied");
  return { ok: errors.length === 0, errors, changes };
}

let cache = null; // { at, result }
export const invalidateOnboardingPolicyCache = () => { cache = null; };

const restrictiveResult = (source, why) => ({
  policy: { ...RESTRICTIVE }, source, revision: 0, updatedAt: null, updatedBy: null, issues: why ? [why] : [], rowToken: null,
});

async function readRow() {
  const { data, error } = await getSupabase()
    .from("system_settings").select("value, updated_at").eq("key", ONBOARDING_POLICY_KEY).maybeSingle();
  if (error) throw error;
  return data || null;
}

function resultFromRow(row) {
  if (!row) {
    return { policy: { ...INITIAL_DEFAULTS }, source: "default", revision: 0, updatedAt: null, updatedBy: null, issues: [], rowToken: null };
  }
  const { policy, issues, meta: m } = normalizePolicyValue(row.value);
  return { policy, source: issues.length ? "invalid" : "database", revision: m.revision, updatedAt: m.updatedAt, updatedBy: m.updatedBy, issues, rowToken: row.updated_at ?? null };
}

// Test seam. Honoured ONLY when NODE_ENV === "test" so it cannot influence a deployed process.
let testOverride = null;
export function __setOnboardingPolicyForTests(partial, source = "database") {
  if (process.env.NODE_ENV !== "test") throw new Error("test seam is only available in NODE_ENV=test");
  testOverride = partial === null ? null : { policy: { ...INITIAL_DEFAULTS, ...partial }, source, revision: 1, updatedAt: null, updatedBy: null, issues: [], rowToken: null };
  invalidateOnboardingPolicyCache();
}

/** Effective policy. Never throws: any failure yields the RESTRICTIVE policy with source "unavailable". */
export async function getOnboardingPolicy({ fresh = false } = {}) {
  if (testOverride && process.env.NODE_ENV === "test") {
    return testOverride.source === "unavailable" ? { ...restrictiveResult("unavailable", "test"), } : testOverride;
  }
  const now = Date.now();
  if (!fresh && cache && now - cache.at < CACHE_TTL_MS) return cache.result;
  if (!isSupabaseConnected()) return restrictiveResult("unavailable", "database not connected");
  try {
    const result = resultFromRow(await readRow());
    cache = { at: now, result };
    return result;
  } catch (err) {
    logError("Onboarding policy read failed — using restrictive policy", { error: err?.message || String(err), code: "ONBOARDING_POLICY_READ_FAILED" });
    return restrictiveResult("unavailable", "read failed");
  }
}

/** True when the policy values may be trusted (row missing is trusted: it is the deliberate default). */
export const isPolicyTrusted = (result) => result?.source === "default" || result?.source === "database";

/**
 * Apply a validated change with compare-and-swap.
 * @returns {{before, after}} effective policies (+ revision)
 * @throws OnboardingPolicyError 400 validation | 409 conflict | 503 store unavailable
 */
export async function updateOnboardingPolicy({ changes: rawChanges, expectedRevision, actorId }) {
  const v = validatePolicyChanges(rawChanges);
  if (!v.ok) throw new OnboardingPolicyError("ONBOARDING_POLICY_INVALID", "Invalid onboarding settings", 400, { errors: v.errors });
  if (!Number.isInteger(expectedRevision) || expectedRevision < 0) {
    throw new OnboardingPolicyError("ONBOARDING_POLICY_REVISION_REQUIRED", "expectedRevision (integer) is required", 400, { errors: ["expectedRevision is required"] });
  }
  if (!isSupabaseConnected()) throw new OnboardingPolicyError("ONBOARDING_POLICY_UNAVAILABLE", "Settings store is unavailable", 503);

  let row;
  try { row = await readRow(); } catch (err) {
    logError("Onboarding policy write aborted: read failed", { error: err?.message || String(err), code: "ONBOARDING_POLICY_READ_FAILED" });
    throw new OnboardingPolicyError("ONBOARDING_POLICY_UNAVAILABLE", "Settings store is unavailable", 503);
  }
  const current = resultFromRow(row);
  if (current.revision !== expectedRevision) {
    throw new OnboardingPolicyError("ONBOARDING_POLICY_CONFLICT", "Settings changed since you loaded them. Reload and review before saving.", 409, { current: publicView(current) });
  }
  if (v.changes.privateSellerApprovalMode === "required" && current.policy.privateSellerApprovalMode !== "required") {
    // Approval mode depends on the persisted capability table. Do not let an administrator
    // enable a gate that the current database cannot enforce or review.
    const capabilityCheck = await getSupabase().from("private_seller_capabilities").select("user_id").limit(1);
    if (capabilityCheck.error) {
      const missing = ["42P01", "PGRST205", "PGRST204"].includes(String(capabilityCheck.error.code || ""))
        || /private_seller_capabilities.*(does not exist|schema cache)/i.test(String(capabilityCheck.error.message || ""));
      if (missing) {
        throw new OnboardingPolicyError("ONBOARDING_POLICY_PREREQUISITE_MISSING", "Private-seller approval cannot be enabled until the capability-table migration has been reconciled and applied.", 503);
      }
      throw new OnboardingPolicyError("ONBOARDING_POLICY_PREREQUISITE_UNAVAILABLE", "Could not verify the private-seller capability table. Settings were not changed.", 503);
    }
  }
  // A corrupt stored policy is repaired from the trusted merge (corrupt fields were already replaced by RESTRICTIVE).
  const nextRevision = current.revision + 1;
  const stamp = new Date().toISOString();
  const value = { schemaVersion: SCHEMA_VERSION, ...current.policy, ...v.changes, revision: nextRevision, updatedAt: stamp, updatedBy: actorId ? String(actorId) : null };

  const sb = getSupabase();
  let write;
  if (!row) {
    write = await sb.from("system_settings").insert({ key: ONBOARDING_POLICY_KEY, value, description: "Owner-controlled onboarding policy (Stage 5)" }).select("key");
    if (write.error) {
      if (write.error.code === "23505") throw new OnboardingPolicyError("ONBOARDING_POLICY_CONFLICT", "Settings were created by someone else. Reload and review.", 409);
      logError("Onboarding policy insert failed", { error: write.error.message, code: "ONBOARDING_POLICY_WRITE_FAILED" });
      throw new OnboardingPolicyError("ONBOARDING_POLICY_WRITE_FAILED", "Could not save settings", 500);
    }
  } else {
    // Compare-and-swap on the stored revision (a JSON-path filter), not on a timestamp: timestamptz text
    // round-trips (precision/offset formatting) are a needless way to get a spurious 409. The UPDATE only
    // matches while the revision we validated is still the stored one; the loser updates zero rows.
    // A stored value with no/zero revision (never written by this service) is matched on its absence of one.
    let q = sb.from("system_settings").update({ value, updated_at: stamp }).eq("key", ONBOARDING_POLICY_KEY);
    q = Number.isInteger(row.value?.revision) ? q.eq("value->>revision", String(row.value.revision)) : q.is("value->>revision", null);
    write = await q.select("key");
    if (write.error) {
      logError("Onboarding policy update failed", { error: write.error.message, code: "ONBOARDING_POLICY_WRITE_FAILED" });
      throw new OnboardingPolicyError("ONBOARDING_POLICY_WRITE_FAILED", "Could not save settings", 500);
    }
  }
  if (!Array.isArray(write.data) || write.data.length !== 1) {
    // Zero rows affected: somebody else won the race between our read and our write.
    throw new OnboardingPolicyError("ONBOARDING_POLICY_CONFLICT", "Settings changed while saving. Reload and review.", 409);
  }
  invalidateOnboardingPolicyCache();
  const after = { policy: { ...current.policy, ...v.changes }, revision: nextRevision, updatedAt: stamp, updatedBy: actorId ? String(actorId) : null, source: "database" };
  return { before: publicView(current), after: publicView(after) };
}

/** Pure: may `user` take an approve/reject decision in `domain` under `result`? */
export function mayDecide(user, domain, result) {
  if (!user) return false;
  const authority = domain === "listing" ? result.policy.listingDecisionAuthority : result.policy.dealerDecisionAuthority;
  // The owner is identified by EMAIL (config/owners.js) and is exempt from the generic email-verification gate.
  // If the owner's account did not exist, anyone could register that address and inherit owner power. So for
  // owner-only authority we additionally require that the owner's own email is VERIFIED.
  const isOwner = user.effectiveRole === "webhoist" && user.emailVerified === true;
  if (authority === "admins") return isOwner || user.role === "admin" || user.role === "superadmin";
  return isOwner; // owner_only (and any untrusted policy): only the verified platform owner
}

/** Express middleware: enforce the owner's decision-authority setting server-side. */
export const requireDecisionAuthority = (domain) => async (req, res, next) => {
  try {
    const result = await getOnboardingPolicy();
    if (mayDecide(req.user, domain, result)) return next();
    logWarn("Decision blocked by owner decision-authority setting", { domain, userId: req.user?.id, role: req.user?.role, source: result.source });
    return res.status(403).json({
      success: false,
      code: "DECISION_AUTHORITY_REQUIRED",
      message: domain === "listing"
        ? "Listing approval and rejection are reserved for the platform owner under the current onboarding settings."
        : "Dealer and seller approval and rejection are reserved for the platform owner under the current onboarding settings.",
    });
  } catch (err) {
    logError("Decision authority check failed — denying", { error: err?.message, code: "DECISION_AUTHORITY_CHECK_FAILED" });
    return res.status(503).json({ success: false, code: "ONBOARDING_POLICY_UNAVAILABLE", message: "Settings are temporarily unavailable. Try again shortly." });
  }
};

/** Effective private-seller state for a user ({id, role, status}). Returns { status, source } where status is
 *  none | pending | approved | rejected | suspended | revoked | not_applicable. */
export const resolveSellerState = async (user) => {
  if (!user) return { status: "none", source: "none" };
  if (user.role === "dealer" || STAFF_ROLES.includes(user.role)) return { status: "not_applicable", source: "role" };

  const policyResult = await getOnboardingPolicy();
  const approvalRequired = policyResult.policy.privateSellerApprovalMode === "required";
  const row = await getCapabilityRow(user.id ?? user._id);

  // Suspension/revocation remain enforceable safety controls even while routine seller approval is disabled.
  if (row && ["suspended", "revoked"].includes(row.status)) {
    return { status: row.status, source: "capability", application: map(row) };
  }
  if (user.role === "individual_seller" && ["suspended", "revoked"].includes(String(user.status || "").toLowerCase())) {
    return { status: String(user.status).toLowerCase(), source: "legacy_users_status" };
  }
  if (!approvalRequired) {
    return { status: "approved", source: "policy_default", application: map(row) };
  }

  if (row) return { status: row.status, source: row.source === "legacy_backfill" ? "legacy_backfill" : "capability", application: map(row) };
  if (user.role === "individual_seller") {
    const legacyStatus = String(user.status || "").toLowerCase();
    if (["suspended", "revoked"].includes(legacyStatus)) return { status: legacyStatus, source: "legacy_users_status" };
    return { status: LEGACY_MAP[user.status] || "pending", source: "legacy_users_status" };
  }
  return { status: "none", source: "none" };
};

  };
};

const PHONE_RE = /^\+?[0-9]{9,15}$/;
export const submitApplication = async ({ user, contactPhone, intendedUse }) => {
  const onboarding = await getOnboardingPolicy();
  if (onboarding.policy.privateSellerApprovalMode !== "required") {
    throw err(409, "SELLER_APPROVAL_NOT_REQUIRED", "Private selling is open. No application or administrator approval is required.");
  }
  if (onboarding.policy.privateSellerPackageMode !== "free") {
    throw err(503, "SELLER_PACKAGE_CONFIGURATION_INCOMPLETE", "Paid seller packages are not available until package checkout is configured.");
  }
  if (user.role !== "user") throw err(403, "SELLER_APPLICATION_NOT_ALLOWED", "Only personal buyer accounts can apply for private-seller access.");
  if (user.emailVerified !== true) throw err(403, "EMAIL_VERIFICATION_REQUIRED", "Please verify your email address before applying to sell.");
  const phone = String(contactPhone || "").replace(/[\s-]/g, "");
  if (!PHONE_RE.test(phone)) throw err(400, "INVALID_PHONE", "A valid contact phone number is required.");
  const note = intendedUse == null ? null : String(intendedUse).trim().slice(0, 500) || null;
  const sb = getSupabase();
  const existing = await getCapabilityRow(user.id);
  if (existing && existing.status !== "rejected") throw err(409, "SELLER_APPLICATION_EXISTS", `A private-seller application already exists (status: ${existing.status}).`);
  const now = new Date().toISOString();
  if (!existing) {
    const { data, error } = await sb.from(TABLE).insert({ user_id: user.id, status: "pending", contact_phone: phone, intended_use: note, source: "application", submitted_at: now }).select("*").single();
    // The primary key makes a concurrent double-submit fail closed instead of creating a second application.
    if (error) { if (error.code === "23505") throw err(409, "SELLER_APPLICATION_EXISTS", "A private-seller application already exists."); throw error; }
    return map(data);
  }
  // Re-application after rejection: only from 'rejected', guarded in the WHERE so a concurrent change cannot be overwritten.
  const { data, error } = await sb.from(TABLE).update({ status: "pending", contact_phone: phone, intended_use: note, submitted_at: now, reviewed_by: null, reviewed_at: null, review_reason: null, updated_at: now })
    .eq("user_id", user.id).eq("status", "rejected").select("*").maybeSingle();
  if (error) throw error;
  if (!data) throw err(409, "SELLER_APPLICATION_EXISTS", "A private-seller application already exists.");
  return map(data);
};

export const decideApplication = async ({ targetUserId, decision, reason, adminId }) => {
  const rule = DECISIONS[decision];
  if (!rule) throw err(400, "INVALID_DECISION", "Unknown decision.");
  const row = await getCapabilityRow(targetUserId);
  if (!row) throw err(404, "SELLER_APPLICATION_NOT_FOUND", "No private-seller application exists for this user.");
  if (!rule.from.includes(row.status)) throw err(409, "INVALID_TRANSITION", `Cannot ${decision} an application that is ${row.status}.`);
  if (["reject", "suspend", "revoke"].includes(decision) && !String(reason || "").trim()) throw err(400, "REASON_REQUIRED", "A reason is required for this decision.");
  const now = new Date().toISOString();
  const { data, error } = await getSupabase().from(TABLE).update({ status: rule.to, reviewed_by: adminId, reviewed_at: now, review_reason: reason ? String(reason).slice(0, 500) : null, updated_at: now })
    .eq("user_id", targetUserId).eq("status", row.status).select("*").maybeSingle();
  if (error) throw error;
  if (!data) throw err(409, "INVALID_TRANSITION", "The application changed while it was being reviewed; reload and retry.");
  // The database visibility trigger hides/restores eligible public listings atomically with the capability
  // transition above (same statement, same transaction). Phase 13F: report the outcome truthfully. Cache invalidation
  // runs in strict mode so a Redis failure is surfaced to the administrator instead of being swallowed; the decision
  // itself is NOT undone because the database (and every public read path's status filter) is authoritative.
  const visibilityChanged = (["suspended", "revoked"].includes(data.status) && row.status !== data.status)
    || (data.status === "approved" && ["suspended", "revoked"].includes(row.status));
  const visibility = { changed: visibilityChanged, cacheInvalidated: null, listingsHiddenByPolicy: null };
  if (visibilityChanged) {
    try {
      const { invalidatePublicListingCaches } = await import("../utils/publicListingCache.js");
      await invalidatePublicListingCaches({ strict: true });
      visibility.cacheInvalidated = true;
    } catch (cacheError) {
      visibility.cacheInvalidated = false;
      console.error("Private-seller listing cache invalidation failed", cacheError?.message || cacheError);
    }
    try {
      const { count, error: countError } = await getSupabase().from(EVENTS_TABLE).select("car_id", { count: "exact", head: true }).eq("seller_user_id", targetUserId).is("restored_at", null);
      if (!countError) visibility.listingsHiddenByPolicy = count ?? 0;
    } catch { /* informational only */ }
  }
  return { before: row.status, after: data.status, application: map(data), visibility };
};

export const listApplications = async ({ status, limit = 50 } = {}) => {
  let q = getSupabase().from(TABLE).select("*").order("submitted_at", { ascending: false }).limit(Math.min(Number(limit) || 50, 200));
  if (status && SELLER_STATUSES.includes(status)) q = q.eq("status", status);
  const { data, error } = await q;
  if (error) throw error;
  return (data || []).map(map);
};

/** Keeps the capability row in step when the LEGACY admin approval endpoint approves an individual_seller.
 *  Only a pending/rejected row is moved to approved. A suspended/revoked row is deliberately NOT reinstated here (reinstatement
 *  restores hidden listings and must go through the review workflow); the returned status lets the caller report that truthfully.
 *  Phase 13F: database errors now propagate (they were silently ignored) and the reviewing administrator is recorded. */
export const syncLegacyApproval = async (userId, adminId = null) => {
  const row = await getCapabilityRow(userId);
  if (!row) return { synced: false, status: null };
  if (row.status === "approved") return { synced: false, status: "approved" };
  if (!["pending", "rejected"].includes(row.status)) return { synced: false, status: row.status };
  const now = new Date().toISOString();
  const { data, error } = await getSupabase().from(TABLE)
    .update({ status: "approved", reviewed_by: adminId, reviewed_at: now, updated_at: now })
    .eq("user_id", userId).in("status", ["pending", "rejected"]).select("status");
  if (error) throw error;
  return { synced: (data || []).length > 0, status: (data || [])[0]?.status || row.status };
};

const call = async (method, path, who, body) => {
  const headers = { "Content-Type": "application/json" };
  if (who) headers.Authorization = `Bearer ${token(who)}`;
  // A request that passes the guards may reach a mocked handler that never answers: recorded as status 0
  // ("guard admitted the caller"), distinct from 401/403 (guard denied).
  const payload = body && method !== "GET" ? JSON.stringify(body) : undefined;
  if (payload) headers["Content-Length"] = Buffer.byteLength(payload);
  return new Promise((resolve) => {
    const u = new URL(base + path);
    const req = http.request({ host: u.hostname, port: u.port, path: u.pathname + u.search, method, headers, agent: false }, (res) => {
      let data = "";
      res.on("data", (c) => { data += c; });
      res.on("end", () => { let json = null; try { json = JSON.parse(data); } catch { /* empty */ } resolve({ status: res.statusCode, json }); });
    });
    const t = setTimeout(() => { req.destroy(); resolve({ status: 0, json: null }); }, 500);
    req.on("close", () => clearTimeout(t));
    req.on("error", () => resolve({ status: 0, json: null }));
    if (payload) req.write(payload);
    req.end();
  });
};

describe("Review lifecycle (authorized, validated, audited)", () => {
  beforeEach(async () => { await call("POST", "/api/private-seller-capability/apply", BUYER, APPLY); audit.mockClear(); });
  const decide = (who, body, id = BUYER) => call("PATCH", `/api/private-seller-capability/admin/${id}/decision`, who, body);
  test("buyers, sellers, dealers and anonymous cannot decide or list", async () => {
    for (const who of [BUYER, BUYER2, LEGACY_APPROVED, DEALER]) {
      expect((await decide(who, { decision: "approve" })).status).toBe(403);
      expect((await call("GET", "/api/private-seller-capability/admin/applications", who)).status).toBe(403);
    }
    expect((await decide(null, { decision: "approve" })).status).toBe(401);
    expect(table.get(BUYER).status).toBe("pending");
  });
  test("a user cannot approve their own application", async () => { expect((await decide(BUYER, { decision: "approve" })).status).toBe(403); expect(table.get(BUYER).status).toBe("pending"); });
  test("admin approves → approved, reviewer recorded, audited; buyer account untouched", async () => {
    const r = await decide(ADMIN, { decision: "approve" });
    expect(r.status).toBe(200); expect(r.json.data).toMatchObject({ before: "pending", after: "approved" });
    expect(table.get(BUYER)).toMatchObject({ status: "approved", reviewed_by: ADMIN });
    expect(audit).toHaveBeenCalledWith(expect.anything(), "private_seller_approve", expect.objectContaining({ target: BUYER }));
    expect(users[BUYER]).toMatchObject({ role: "user", status: "approved" });
  });
  test("reject/suspend/revoke need a reason; invalid transitions are 409", async () => {
    expect((await decide(ADMIN, { decision: "reject" })).json.code).toBe("REASON_REQUIRED");
    expect((await decide(ADMIN, { decision: "suspend", reason: "x" })).json.code).toBe("INVALID_TRANSITION"); // pending cannot be suspended
    await decide(ADMIN, { decision: "approve" });
    expect((await decide(ADMIN, { decision: "suspend", reason: "policy breach" })).json.data.after).toBe("suspended");
    expect((await decide(ADMIN, { decision: "approve" })).json.data.after).toBe("approved"); // reinstated
    expect((await decide(ADMIN, { decision: "revoke", reason: "fraud" })).json.data.after).toBe("revoked");
    expect((await decide(ADMIN, { decision: "reject", reason: "late" })).json.code).toBe("INVALID_TRANSITION");
  });
  test("unknown decision / malformed id / no application", async () => {
    expect((await decide(ADMIN, { decision: "promote" })).status).toBe(400);
    expect((await decide(ADMIN, { decision: "approve" }, "not-a-uuid")).status).toBe(400);
    expect((await decide(ADMIN, { decision: "approve" }, BUYER2)).json.code).toBe("SELLER_APPLICATION_NOT_FOUND");
  });
  test("admin list returns applications", async () => {
    const r = await call("GET", "/api/private-seller-capability/admin/applications?status=pending", ADMIN);
    expect(r.status).toBe(200); expect(r.json.data.map((a) => a.userId)).toEqual([BUYER]);
  });
  test("legacy admin approval synchronization updates only an existing pending/rejected capability row", async () => {
    table.set(BUYER, { user_id: BUYER, status: "pending", source: "application", reviewed_at: null });
    await svc.syncLegacyApproval(BUYER);
    expect(table.get(BUYER).status).toBe("approved");
    expect(table.get(BUYER).reviewed_at).toBeTruthy();
    table.set(BUYER2, { user_id: BUYER2, status: "suspended", source: "application" });
    await svc.syncLegacyApproval(BUYER2);
    expect(table.get(BUYER2).status).toBe("suspended");
  });
});

describe("Listing guard (sellerListingAccess): capability enforced at the backend", () => {
  const run = (user, opts) => new Promise((resolve) => {
    const req = { user: { ...user, id: user.id || user._id } };
    const res = { status(c) { this.c = c; return this; }, json(b) { resolve({ status: this.c, body: b, req }); } };
    sellerListingAccess(opts)(req, res, (e) => resolve({ passed: true, error: e, req }));
  });
  const seed = (id, status) => table.set(id, { user_id: id, status, source: "application" });
  test.each([["none", null, false], ["pending", "pending", false], ["rejected", "rejected", false], ["suspended", "suspended", false], ["revoked", "revoked", false]])("buyer (%s) cannot write listings (direct API)", async (_n, st) => {
    if (st) seed(BUYER, st);
    const r = await run(users[BUYER], { write: true });
    expect(r.passed).toBeUndefined(); expect(r.status).toBe(403); expect(r.body.code).toBe("SELLER_CAPABILITY_REQUIRED");
  });
  test("approved buyer-account seller may write; req.sellerCapability is set; no dealer privilege", async () => {
    seed(BUYER, "approved");
    const r = await run(users[BUYER], { write: true });
    expect(r.passed).toBe(true); expect(r.req.sellerCapability.status).toBe("approved");
    const dealerGuard = await new Promise((resolve) => { const res = { status(c) { this.c = c; return this; }, json() { resolve(this.c); } }; dealerBusinessOnly({ user: users[BUYER] }, res, () => resolve("passed")); });
    expect(dealerGuard).toBe(403);
  });
  test("unverified email cannot sell even when approved", async () => { seed(UNVERIFIED, "approved"); expect((await run(users[UNVERIFIED], { write: true })).status).toBe(403); });
  test("seller views (write:false) require approved capability; applications remain on their separate status endpoint", async () => {
    expect((await run(users[BUYER], { write: false })).status).toBe(403);
    for (const st of ["pending", "rejected", "suspended", "revoked"]) { seed(BUYER, st); expect((await run(users[BUYER], { write: false })).status).toBe(403); }
    seed(BUYER, "approved");
    expect((await run(users[BUYER], { write: false })).passed).toBe(true);
  });
  test("suspension/revocation blocks seller writes only; buyer reads of the account are not this guard's concern", async () => {
    seed(BUYER, "suspended"); expect((await run(users[BUYER], { write: true })).status).toBe(403);
    seed(BUYER, "revoked"); expect((await run(users[BUYER], { write: true })).status).toBe(403);
  });
  test("legacy individual_seller: approved works; pending is blocked when approval is required; suspended/revoked remain blocked", async () => {
    expect((await run(users[LEGACY_APPROVED], { write: true })).passed).toBe(true);
    expect((await run(users[LEGACY_PENDING], { write: true })).status).toBe(403);
    const s = await run(users[LEGACY_SUSP], { write: true }); expect(s.status).toBe(403); expect(s.body.code).toBe("SELLER_CAPABILITY_BLOCKED");
    expect((await run(users[LEGACY_SUSP], { write: false })).status).toBe(403);
    table.set(LEGACY_APPROVED, { user_id: LEGACY_APPROVED, status: "revoked", source: "legacy_backfill" });
    expect((await run(users[LEGACY_APPROVED], { write: true })).status).toBe(403); // the capability row is authoritative once present
  });
  test("dealers and staff keep dealerOnly behaviour", async () => {
    expect((await run(users[DEALER], { write: true })).passed).toBe(true);
    expect((await run(users[ADMIN], { write: true })).passed).toBe(true);
    expect((await run({ ...users[SUPPORT], role: "buyer_x" }, { write: true })).status).toBe(403);
  });
  test("a seller capability never lets one user act on another's capability (state is per users.id)", async () => {
    seed(BUYER, "approved");
    expect((await run(users[BUYER2], { write: true })).status).toBe(403);
  });
});
